% Define parameters
N = 31;    % Total number of samples
n = 0:N-1; % Sample index
n1 = 10;   % Start of rectangular pulse
n2 = 20;   % End of rectangular pulse

% Define the rectangular pulse x[n]
x = zeros(1, N);   % Initialize the signal with zeros
x(n1+1:n2+1) = 1; % Set values between n1 and n2 to 1 (MATLAB uses 1-based indexing)

% Compute DTFT
omega = linspace(-pi, pi, 1000); % Frequency range 
X = x * exp(-1i * n' * omega);   % DTFT calculation via matrix multiplication

% --- Plot 1: Magnitude of DTFT ---
figure(1); clf;
plot(omega/pi, abs(X), 'b', 'LineWidth', 1.2);
xlabel('Normalized Frequency (\omega/\pi)'); 
ylabel('|X(e^{j\omega})|');
title('Magnitude of DTFT of Rectangular Pulse');
xlim([-1 1]); 
ylim([0 12]);
xticks(-1:0.2:1);
grid on;

% --- Plot 2: Phase of DTFT ---
figure(2); clf;
plot(omega/pi, atan(imag(X)./real(X)), 'b', 'LineWidth', 1.2); 
xlabel('Normalized Frequency (\omega/\pi)'); 
ylabel('Phase of X(e^{j\omega})'); 
title('Phase of DTFT of Rectangular Pulse');
xlim([-1 1]); 
ylim([-2 2]);
xticks(-1:0.2:1);
grid on;
