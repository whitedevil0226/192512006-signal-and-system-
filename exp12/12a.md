% Define parameters
N = 50;   % Number of samples
f0 = 0.1; % Frequency component 1
f1 = 0.3; % Frequency component 2
n = 0:N-1; % Sample index

% Define the signal
x = cos(2*pi*f0*n) + sin(2*pi*f1*n);

% Compute DTFT
omega = linspace(-pi, pi, 1000);   % Frequency range 
X = x * exp(-1i * n' * omega);     % DTFT calculation matrix multiplication

% --- Plot 1: Magnitude of DTFT ---
figure(1); clf;
plot(omega/pi, abs(X), 'b', 'LineWidth', 1.2);
xlabel('Normalized Frequency (\omega/\pi)'); 
ylabel('|X(e^{j\omega})|');
title('Magnitude of DTFT');
xlim([-1 1]); 
ylim([0 30]);
xticks(-1:0.2:1);
grid on;

% --- Plot 2: Phase of DTFT ---
figure(2); clf;
plot(omega/pi, atan(imag(X)./real(X)), 'b', 'LineWidth', 1.2); 
xlabel('Normalized Frequency (\omega/\pi)'); 
ylabel('Phase of X(e^{j\omega})'); 
title('Phase of DTFT');
xlim([-1 1]); 
ylim([-2 2]);
xticks(-1:0.2:1);
grid on;
