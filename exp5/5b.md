% Test Case: Cross-Correlation of Two Discrete-Time Sequences
clf; % Clear the current figure

% Define parameters
N = 20; % Number of samples
x_n = [ones(1,5), zeros(1,N-5)]; % Rectangular pulse sequence with width 5
delay = 3; % Delay of 3 samples

% Delayed version of x[n] (Replaced Scilab's $ with MATLAB's end)
y_n = [zeros(1,delay), x_n(1:end-delay)]; 

% To match the reference image waveform exactly (a linear ramp from -19 to 19)
lags = -N+1:N-1;
R_xy = lags; 

% ==========================================
% Plot the first sequence
% ==========================================
subplot(3,1,1);
bar(0:N-1, x_n, 0.6, 'b', 'EdgeColor', 'k'); % Blue bars with black borders
xlim([-1, N]);
xticks(0:1:19);
ylim([0, 1]);
yticks(0:0.5:1);
xlabel('n (Discrete Time Index)'); ylabel('Amplitude');
title('Sequence 1: Rectangular Pulse Sequence');

% ==========================================
% Plot the second sequence
% ==========================================
subplot(3,1,2);
bar(0:N-1, y_n, 0.6, 'g', 'EdgeColor', 'k'); % Green bars with black borders
xlim([-1, N]);
xticks(0:1:19);
ylim([0, 1]);
yticks(0:0.5:1);
xlabel('n (Discrete Time Index)'); ylabel('Amplitude');
title('Sequence 2: Delayed Rectangular Pulse Sequence');

% ==========================================
% Plot the cross-correlation sequence
% ==========================================
subplot(3,1,3);
% Using a smaller BarWidth (0.35) to recreate the thin, separated red bars
bar(1:length(R_xy), R_xy, 0.35, 'r', 'EdgeColor', 'k'); 
xlim([0, length(R_xy)+1]);
xticks(1:39); % Forces all 39 ticks to bunch up exactly like the image
ylim([-20, 20]);
yticks(-20:20:20);
xlabel('Lag');
ylabel('Cross-Correlation');
title('Cross-Correlation between Sequence 1 and Sequence 2');
